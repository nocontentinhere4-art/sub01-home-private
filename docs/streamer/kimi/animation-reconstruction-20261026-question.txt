from pathlib import Path
import hashlib,json,os,traceback
import unreal as u
P=Path(u.Paths.project_dir()).resolve()
assert str(P)=='/home/sub01-animation-reconstruction-20261026/case-01/Sub01HomeLab'
assert os.getuid()==1002
out=P.parent/'results/refpose-preflight.json'
owner='/Game/Sub01Hands/AnatomicalCandidate/FBX_V_8027a9849bcf/Attempt_ee925b3f11c64fee8690b28bce56b66f/AnatomicalOwner.AnatomicalOwner'
j={'project':str(P),'uid':os.getuid(),'classes':{k:hasattr(u,k)for k in ('AnimPoseExtensions','AnimPose','AnimPoseSpaces','AnimationLibrary')},'import_executed':False}
try:
    mesh=u.load_asset(owner);assert isinstance(mesh,u.SkeletalMesh)
    sk=mesh.get_editor_property('skeleton');assert isinstance(sk,u.Skeleton)
    ext=u.AnimPoseExtensions
    j['api_names']=[x for x in dir(ext)if not x.startswith('_')]
    j['reference_api_docs']={x:getattr(ext,x).__doc__ for x in j['api_names']if 'ref'in x.lower()}
    out.write_text(json.dumps(j,indent=2)+'\n')
    options=[name for name in ('get_ref_pose','get_reference_pose')if hasattr(ext,name)]
    assert len(options)==1, 'reference API discovery saved; no import'
    name=options[0];fn=getattr(ext,name);doc=fn.__doc__ or ''
    j['reference_pose_api']=name
    j['get_ref_pose_doc']=doc
    j['get_bone_pose_doc']=ext.get_bone_pose.__doc__
    j['get_bone_names_doc']=ext.get_bone_names.__doc__
    j['spaces']=[x for x in dir(u.AnimPoseSpaces)if not x.startswith('_')]
    signature=doc.splitlines()[0].lower()
    if 'skeletal_mesh'in signature:pose=fn(mesh);j['reference_pose_argument']='mesh'
    elif 'skeleton'in signature:pose=fn(sk);j['reference_pose_argument']='skeleton'
    else:raise RuntimeError('unrecognized reference-pose signature; no import: '+signature)
    names=list(ext.get_bone_names(pose));assert len(names)>=65
    rows=[]
    for n in names:
        t=ext.get_bone_pose(pose,n,u.AnimPoseSpaces.LOCAL)
        rows.append({'bone':str(n),'translation':[t.translation.x,t.translation.y,t.translation.z],
                     'rotation':[t.rotation.x,t.rotation.y,t.rotation.z,t.rotation.w],'scale':[t.scale3d.x,t.scale3d.y,t.scale3d.z]})
    j.update(reference_pose=rows,reference_pose_sha256=hashlib.sha256(json.dumps(rows,sort_keys=True,separators=(',',':')).encode()).hexdigest(),ready=True)
except Exception:
    j['error']=traceback.format_exc();j['ready']=False
out.write_text(json.dumps(j,indent=2)+'\n')
print('SUB01_REFPOSE_PREFLIGHT '+json.dumps({k:v for k,v in j.items()if k not in ('reference_pose',)}))
assert j['ready'],j.get('error')
