def angle(a, b):
    aa = [a.x, a.y, a.z, a.w]
    bb = [b.x, b.y, b.z, b.w]
    cosine = abs(sum(x*y for x, y in zip(aa, bb))) / math.sqrt(sum(x*x for x in aa) * sum(x*x for x in bb))
    return math.degrees(2 * math.acos(min(1, max(0, cosine))))
