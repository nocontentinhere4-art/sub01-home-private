def inspect_frames(directory,reference,window):
    """Independent on-disk journal/frame readback; safe only after collector closes."""
    directory=Path(directory); journal=directory/'frames.jsonl'
    rows=[json.loads(x) for x in journal.read_text().splitlines()]
    need(rows,'No captured frames'); seen=set();inodes=set();previous=None;last_time=None
    for ordinal,r in enumerate(rows,1):
        need(r['ordinal']==ordinal,'Journal ordinal gap')
        name=r['saved_frame'];need(name==f'frame-{ordinal:06d}-seq{r["seq"]}.jpg','Unexpected frame name')
        need(name not in seen,'Repeated saved identity');seen.add(name);p=directory/name;s=p.lstat()
        need(stat.S_ISREG(s.st_mode) and s.st_nlink==1,'Non-independent frame file')
        need((s.st_dev,s.st_ino)not in inodes,'Shared frame inode');inodes.add((s.st_dev,s.st_ino))
        b=p.read_bytes();need(len(b)==r['jpeg_bytes'] and sha_bytes(b)==r['sha256'],'Frame readback mismatch')
        ok,seq,why=reference.check_contract(b);need(ok and seq==r['seq'],'Readback JPEG/seq invalid: '+why)
        if previous is not None:need(seq==previous+1,'Unexplained sequence gap/regression')
        previous=seq;t=r['arrival_monotonic_ns']
        need(window['start_ns']<=t<window['deadline_ns'],'Frame outside capture window')
        need(last_time is None or t>=last_time,'Arrival time regression');last_time=t
    need({x.name for x in directory.glob('*.jpg')}==seen,'Extra/unlisted JPEG')
    need(len(rows)==window['frames'] and window['save_skipped']==0,'Incomplete saved frames')
    need(window['end_ns']>=window['deadline_ns'],'Early capture termination')
    need(window['deadline_ns']-window['start_ns']==15_000_000_000,'Not full15s capture')
    # byte bound includes ALL FIFO read bytes (including pre-window) + journal bytes.
    need(journal.stat().st_size==window['journal_bytes'],'Journal byte accounting mismatch')
    need(sum(r['jpeg_bytes']for r in rows)<=window['input_bytes'],'Saved JPEG bytes exceed FIFO input')
    need(window['input_bytes']+journal.stat().st_size<512*1024*1024,'Byte bound reached')
    need(len(rows)<1000,'Frame bound reached')
    arrivals=[r['arrival_monotonic_ns'] for r in rows]
    gaps=[arrivals[0]-window['start_ns'],window['deadline_ns']-arrivals[-1]]+[b-a for a,b in zip(arrivals,arrivals[1:])]
    need(max(gaps)<=window['max_gap_ns'],'Unexplained leading/interframe/trailing timing gap')
    return {'frames':len(rows),'input_plus_journal_bytes':window['input_bytes']+journal.stat().st_size,'max_gap_ns':max(gaps),'journal_sha256':sha_bytes(journal.read_bytes()),'status':'READBACK_PASS'}

class FrameStore:
    """Synchronous complete frame saving; bounds fail, never truncate as success."""
    def __init__(self,directory,reference,max_frames=1000,max_bytes=512*1024*1024):
        self.directory=Path(directory);self.directory.mkdir(mode=0o700);self.ref=reference
        self.dirfd=os.open(self.directory,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC)
        self.journal=open(self.directory/'frames.jsonl','xb',buffering=0)
        self.frames=0;self.input_bytes=0;self.journal_bytes=0;self.last_seq=None;self.max_frames=max_frames;self.max_bytes=max_bytes
    def account(self,n):
        need(n>=0,'Negative input accounting');self.input_bytes+=n
        need(self.input_bytes+self.journal_bytes<self.max_bytes,'BYTE_BOUND_REACHED')
    def append(self,frame,arrival_ns,read_ordinal):
        ok,seq,why=self.ref.check_contract(frame);need(ok and seq is not None and seq>0,'JPEG_CONTRACT: '+why)
        need(self.last_seq is None or seq==self.last_seq+1,'SEQUENCE_GAP_OR_REGRESSION')
        ordinal=self.frames+1;need(ordinal<self.max_frames,'FRAME_BOUND_REACHED')
        name=f'frame-{ordinal:06d}-seq{seq}.jpg'
        row={'ordinal':ordinal,'seq':seq,'arrival_monotonic_ns':arrival_ns,'read_ordinal':read_ordinal,'jpeg_bytes':len(frame),'sha256':sha_bytes(frame),'saved_frame':name}
        b=(json.dumps(row,sort_keys=True)+'\n').encode();need(self.input_bytes+self.journal_bytes+len(b)<self.max_bytes,'BYTE_BOUND_REACHED')
        fd=os.open(name,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW|os.O_CLOEXEC,0o600,dir_fd=self.dirfd)
        with os.fdopen(fd,'wb')as f:f.write(frame);f.flush();os.fsync(f.fileno())
        self.journal.write(b);self.journal_bytes+=len(b);self.frames=ordinal;self.last_seq=seq
    def close(self):
        if not self.journal.closed:os.fsync(self.journal.fileno());self.journal.close()
        if self.dirfd is not None:os.close(self.dirfd);self.dirfd=None

class Observer:
    def __init__(self,fifo_path,evidence_dir,approved_background,*,executable_pins,launch_floor_ticks):
        """approved_background: explicit reviewed [{'identity':..., 'reason':...}].
        Must include every initial process other than this supervisor; never auto-approved.
        """
        flag=ctypes.c_int();need(ctypes.CDLL(None,use_errno=True).prctl(37,ctypes.byref(flag),0,0,0)==0 and flag.value==1,'Supervisor subreaper not enabled')
        self.executable_pins=executable_pins;self.launch_floor_ticks=launch_floor_ticks
        self.dynamic_executables={p for p in executable_pins if Path(p).name in {'UnrealEditor-Cmd','UnrealTraceServer','zenserver'}}
        self.network_baseline=None
        self.ancestry={};self.terminal_records={};self.lifecycle_failures=[]
        global DIAGNOSTIC_SINK
        DIAGNOSTIC_SINK=lambda event:self.event("inspection-diagnostic",diagnostic=event)
        self.path=Path(fifo_path);self.out=Path(evidence_dir);self.out.mkdir(mode=0o700)
        self.background={int(x['identity']['pid']):x for x in approved_background}
        need(len(self.background)==len(approved_background),'Duplicate background approval')
        for x in approved_background:need(bool(x.get('reason')),'Background behavior rationale required')
        self.owned={os.getpid():{'identity':identity(os.getpid()),'role':'observer-supervisor'}}
        self.setup_phase='UNTOUCHED';self.target=None;self.old=None;self.warm=None;self.event_count=0;self.handoff_pass=False
    def event(self,label,**data):
        self.event_count+=1;write_new(self.out/f'{self.event_count:04d}-{label}.json',{'label':label,'monotonic_ns':time.monotonic_ns(),**data})
    def verify_exe(self,row,dynamic=False):
        exe=row['exe'];pin=self.executable_pins.get(exe)
        need(pin is not None and row['exe_sha256']==pin['sha256'],'Executable identity/hash mismatch: '+exe)
        if dynamic:need(exe in self.dynamic_executables,'Unexpected dynamic descendant executable')
    def lifecycle_failure(self,why,**data):
        self.lifecycle_failures.append({'reason':why,**data})
        try:self.event('lifecycle-failure',reason=why,**data)
        except BaseException:pass
    def record_launch(self,pid,ticks,role,transitional_executable=None):
        i=identity(pid);need(i['ticks']==ticks and i['ppid']==os.getpid(),'Direct launch identity/parent mismatch')
        if pid in self.ancestry:need(self.ancestry[pid]['identity']['ticks']==ticks,'Direct launch receipt PID reused')
        need(pid not in self.background,'Background cannot become launch')
        proof={'identity':i,'role':role,'direct_launch':True,'parent_identity':self.owned[os.getpid()]['identity']}
        if transitional_executable is not None and set(i['uids'])=={'0'}:
            need(role=='warm-reader' and transitional_executable=='/usr/bin/setpriv' and i['sid']==pid,'Unapproved root transition launch')
            parent=proof['parent_identity']
            need(all(i[k]==parent[k] for k in ('uids','gids','groups','cap_eff','cap_prm','cap_amb')) and i['no_new_privs'] in (0,1),'Unexpected root transition credentials')
            row=executable_one(pid);need(stable(i,row['identity']) and row['exe']==transitional_executable,'Root transition executable mismatch');self.verify_exe(row)
            binary=Path(transitional_executable);st=binary.stat()
            need(stat.S_ISREG(st.st_mode) and st.st_uid==0 and not(st.st_mode&0o022),'Transition executable not root-controlled')
            need(sha_bytes(binary.read_bytes())==row['exe_sha256'],'Transition executable bytes differ')
            proof['root_transition']={'exe':row['exe'],'exe_sha256':row['exe_sha256'],'identity':i}
        self.ancestry[pid]=proof
    def cleanup_binding(self,pid,current):
        proof=self.ancestry.get(pid) or self.owned.get(pid);need(proof is not None,'Cleanup has no launch proof')
        expected=proof['identity'];need(current['ticks']==expected['ticks'],'Cleanup PID reused')
        transition=proof.get('root_transition')
        if transition:
            self.lifecycle_failure('Unverified root-transition launch cleanup only',pid=pid)
            need(proof.get('direct_launch') and current['ppid']==os.getpid() and current['sid']==pid,'Root transition parent/session drift')
            if set(current['uids'])=={'0'}:
                bound=dict(expected,no_new_privs=current['no_new_privs'])
                need(current['no_new_privs'] in (expected['no_new_privs'],1) and stable(bound,current),'Root transition credential drift')
                if current['state']!='Z':
                    row=executable_one(pid);need(stable(current,row['identity']) and row['exe']==transition['exe'] and row['exe_sha256']==transition['exe_sha256'],'Root transition executable drift');self.verify_exe(row)
                return bound
            require_unprivileged(current)
            need(all(current[k]==expected[k] for k in ('pid','ticks','ppid','sid')),'Transition UID1001 identity drift')
            if current['state']!='Z':
                row=inspect_one(pid,executable_only=True);need(stable(current,row['identity']),'Transition cleanup inspection drift')
                if row['exe']==transition['exe']:need(row['exe_sha256']==transition['exe_sha256'],'Transition binary changed')
                else:
                    self.verify_exe(row);need(row['exe']=='/usr/bin/python3.12','Unapproved warm transition executable')
            # Cleanup-only credentials may finish dropping after the last observation.
            # This never sets owned/executable admission and remains a failed run.
            return current
        bound=expected
        if current['ppid']!=expected['ppid']:
            need(current['ppid']==os.getpid() and proof.get('parent_identity') is not None,'Cleanup adoption lacks earlier chain')
            bound=dict(expected,ppid=os.getpid())
        require_unprivileged(current);need(stable(bound,current),'Cleanup identity drift')
        return bound
    def terminal(self,pid):
        proof=self.ancestry.get(pid) or self.owned.get(pid)
        need(proof is not None,'Terminal ancestry unproven')
        expected=proof['identity'];verified=pid in self.owned and self.owned[pid].get('exe') is not None
        try:
            a=identity(pid);need(a['ticks']==expected['ticks'],'Terminal PID reused');b=identity(pid)
        except FileNotFoundError:
            need(not Path('/proc',str(pid)).exists(),'Partial terminal identity read')
            result={'pid':pid,'identity':expected,'state':'GONE','live_verified':verified}
            if not verified:self.lifecycle_failure('Gone before live verification',pid=pid)
            self.terminal_records[pid]=result;return result
        need(a['ticks']==expected['ticks'],'Terminal PID reused')
        bound=self.cleanup_binding(pid,a)
        need(stable(bound,a) and stable(a,b) and a['state']==b['state']=='Z','Terminal identity not stable Z')
        result={'pid':pid,'identity':b,'state':'VERIFIED_TERMINAL' if verified else 'UNVERIFIED_TERMINAL_CHILD','live_verified':verified,'prior_identity':expected,'parent_identity':proof.get('parent_identity')}
        self.terminal_records[pid]=result
        if not verified:self.lifecycle_failure('Terminal before live executable verification',pid=pid)
        try:self.event('terminal',terminal=result)
        except BaseException:pass
        return result
    def reap_terminal(self,record):
        pid=record['pid'];need(record['state'] in ('VERIFIED_TERMINAL','UNVERIFIED_TERMINAL_CHILD'),'Not a terminal wait candidate')
        current=self.terminal(pid);need(current['state'] in ('VERIFIED_TERMINAL','UNVERIFIED_TERMINAL_CHILD'),'Gone is not reaped')
        need(current['identity']['ppid']==os.getpid(),'Terminal not actual waitable child')
        try:waited,status=os.waitpid(pid,os.WNOHANG)
        except ChildProcessError:
            self.lifecycle_failure('ECHILD is not REAPED',pid=pid);raise Block('Exact child wait returned ECHILD')
        need(waited==pid,'Exact child not reaped')
        need(not Path('/proc',str(pid)).exists(),'PID remains/reappeared after wait; no old-identity reuse')
        result=dict(current,state='REAPED',wait_status=status,returncode=os.waitstatus_to_exitcode(status))
        self.terminal_records[pid]=result
        try:self.event('terminal-reaped',terminal=result)
        except BaseException:pass
        return result
    def lifecycle_cleanup_pass(self,sig):
        outcomes=[];errors=[];pending=[]
        records=dict(self.ancestry);records.update({p:r for p,r in self.owned.items() if p!=os.getpid()})
        for pid,proof in records.items():
            try:
                try:i=identity(pid)
                except FileNotFoundError:
                    need(not Path('/proc',str(pid)).exists(),'Partial identity in cleanup')
                    prior=self.terminal_records.get(pid)
                    if prior and prior['state']=='REAPED':outcomes.append(prior)
                    else:outcomes.append(self.terminal(pid))
                    continue
                expected=proof['identity'];need(i['ticks']==expected['ticks'],'Cleanup PID reused')
                if i['state']=='Z':
                    t=self.terminal(pid)
                    if t['identity']['ppid']==os.getpid():outcomes.append(self.reap_terminal(t))
                    else:pending.append(pid)
                    continue
                bound=self.cleanup_binding(pid,i)
                # A recorded direct Popen identity permits failure-path termination,
                # not live executable admission. Unknown adoption is never recorded.
                need(pid in self.owned or proof.get('direct_launch'),'Unverified live descendant cannot bypass inspection')
                need(stable(i,identity(pid)),'Cleanup signal identity changed');os.kill(pid,sig);pending.append(pid)
            except BaseException as e:
                errors.append({'pid':pid,'error':repr(e)});self.lifecycle_failure('Cleanup identity/wait failure',pid=pid,error=repr(e))
        return {'pending':pending,'errors':errors,'outcomes':outcomes}
    def register_owned(self,pid,start_ticks,role):
        if pid not in self.ancestry:self.record_launch(pid,start_ticks,role)
        try:row=inspect_one(pid,executable_only=True)
        except BaseException as e:
            self.lifecycle_failure('Direct launch executable unverified',pid=pid,error=repr(e))
            raise
        i=row['identity'];need(i['ticks']==start_ticks,'Owned PID reused')
        need(pid not in self.background,'Background cannot silently become owned')
        need(i['ppid']==os.getpid(),'Direct launch relationship absent')
        need(set(i['uids'])=={'1001'} and set(i['gids'])=={'1001'} and i['ticks']>=self.launch_floor_ticks,'Owned direct launch eligibility mismatch')
        require_unprivileged(i);self.verify_exe(row)
        need(Path(row['exe']).name in {'UnrealEditor-Cmd','Xvfb','python3.12'},'Unexpected direct executable')
        self.owned[pid]={'identity':i,'role':role,'exe':row['exe'],'exe_sha256':row['exe_sha256'],'launch_parent':os.getpid()};self.event('owned',identity=i,role=role,exe=row['exe'],exe_sha256=row['exe_sha256'])
    def register_descendants(self):
        try:return self._register_descendants()
        except BaseException as e:
            self.lifecycle_failure('Discovery failed',error=repr(e));raise
    def _register_descendants(self):
        # Actual ancestry / verified subreaper adoption is the ownership proof.
        rows=[]
        for p in Path('/proc').iterdir():
            if p.name.isdigit():
                try:rows.append(identity(int(p.name)))
                except FileNotFoundError:need(not p.exists(),'Transient live identity')
        self.reconcile_descendants(rows)
    def reconcile_descendants(self,rows):
        known={pid for pid,x in self.owned.items() if any(i['pid']==pid and i['ticks']==x['identity']['ticks']for i in rows)}
        changed=True
        while changed:
            changed=False
            for i in rows:
                pid=i['pid']
                if pid==os.getpid() or pid in self.background:continue
                if pid in self.owned:
                    require_unprivileged(i);prev=self.owned[pid]['identity'];need(prev['ticks']==i['ticks'],'Owned PID reused')
                    if i['state']=='Z':self.terminal(pid);continue
                    if prev['ppid']!=i['ppid']:
                        need(i['ppid']==os.getpid(),'Owned process reparented outside supervisor')
                        need((self.ancestry.get(pid) or {}).get('parent_identity') is not None,'Owned adoption lacks recorded parent identity')
                        altered=dict(prev,ppid=i['ppid']);need(stable(altered,i),'Adopted identity/credentials/session drift')
                        row=inspect_one(pid,executable_only=True);self.verify_exe(row,dynamic=True)
                        self.owned[pid]['identity']=i;self.event('owned-reparent',old_parent=prev['ppid'],identity=i)
                    continue
                if i['ppid']in known:
                    need(i['ppid']!=os.getpid() or pid in self.ancestry,'First-seen adoption has no prior chain/direct launch')
                    parent=self.owned[i['ppid']]['identity']
                    need(any(stable(parent,x) for x in rows if x['pid']==i['ppid']),'Verified parent identity changed')
                    self.ancestry.setdefault(pid,{'identity':i,'role':'descendant-observation','parent_identity':parent,'direct_launch':False})
                    require_unprivileged(i)
                    if i['state']=='Z':
                        self.terminal(pid);raise Block('Unverified terminal descendant; cleanup only')
                    need(set(i['uids'])=={'1001'} and set(i['gids'])=={'1001'} and i['ticks']>=self.launch_floor_ticks,'Unknown descendant eligibility failed')
                    try:
                        row=inspect_one(pid,executable_only=True);need(stable(i,row['identity']),'Descendant identity raced');self.verify_exe(row,dynamic=True)
                    except BaseException as e:
                        self.lifecycle_failure('Descendant live verification failed',pid=pid,error=repr(e));raise
                    self.owned[pid]={'identity':i,'role':'owned-descendant','exe':row['exe'],'exe_sha256':row['exe_sha256'],'launch_parent':i['ppid']}
                    self.event('descendant-or-subreaper-adoption',identity=i,actual_parent=i['ppid'],exe=row['exe'],exe_sha256=row['exe_sha256']);known.add(pid);changed=True
    def network_snapshot(self,rows):
        # Only known transport sockets may remain attached to background services.
        background_sockets={str(f['ino']) for r in rows if r['identity']['pid']in self.background for f in r['fds']if f.get('socket')}
        owned_sockets={str(f['ino']) for r in rows if r['identity']['pid']in self.owned for f in r['fds']if f.get('socket')}
        records=set()
        for table in ('tcp','tcp6','udp','udp6'):
            for line in (Path('/proc/net')/table).read_text().splitlines()[1:]:
                fields=line.split();need(len(fields)>=10,'Malformed network inventory')
                # TCP established/listening; all UDP endpoints can carry new control traffic.
                if table.startswith('tcp') and fields[3]not in ('01','0A'):continue
                inode=fields[9];record=(table,fields[1],fields[2],fields[3],inode)
                if inode in owned_sockets and inode not in background_sockets:continue
                records.add(record)
        snapshot={'background_socket_inodes':sorted(background_sockets),'control_network_rows':sorted(records)}
        if self.network_baseline is None:self.network_baseline=snapshot;self.event('network-admission',snapshot=snapshot)
        else:need(snapshot==self.network_baseline,'Background/control connection or listener changed')
    def corroborate(self,label,expected_reader_pids=(),writer_identity=None):
        with FD_LOCK:
            self.register_descendants();rows=census()
        seen=set();readers=[];writers=[]
        for row in rows:
            i=row['identity'];pid=i['pid'];seen.add(pid)
            approved=self.owned.get(pid)or self.background.get(pid)
            need(approved is not None,'Unapproved process: '+str(pid))
            need(stable(approved['identity'],i),'Process identity/credential/parent/session drift: '+str(pid))
            self.verify_exe(row)
            if pid in self.owned and pid!=os.getpid():
                require_unprivileged(i)
                need(row['exe']==approved['exe'] and row['exe_sha256']==approved['exe_sha256'],'Owned executable drift')
            if pid in self.background:
                need(approved['identity'].get('cmdline_sha256')==i['cmdline_sha256'] and approved['identity'].get('comm')==i['comm'],'Background executable/argv identity changed')
            for fd in row['fds']:
                if self.target and (fd['dev'],fd['ino'])==self.target:
                    mode=fd['flags']&os.O_ACCMODE
                    if mode in (os.O_RDONLY,os.O_RDWR):readers.append(pid)
                    if mode in (os.O_WRONLY,os.O_RDWR):
                        need(pid in self.owned,'Unowned/background FIFO writer');writers.append(pid)
        need(set(self.background)<=seen,'Fixed background process disappeared')
        self.network_snapshot(rows)
        need(set(readers)==set(expected_reader_pids),'Unexpected/missing FIFO reader')
        if writer_identity is not None:
            now=identity(writer_identity['pid']);need(now['ticks']==writer_identity['ticks'],'Writer PID reused')
            need(set(writers)<={writer_identity['pid']},'Foreign FIFO writer')
        if self.target:self.check_inode()
        self.event(label,processes=rows,readers=readers,writers=writers)
        return rows
    def check_inode(self):
        s=self.path.lstat();need(stat.S_ISFIFO(s.st_mode) and (s.st_dev,s.st_ino)==self.target and s.st_uid==1001 and stat.S_IMODE(s.st_mode)==0o600,'FIFO identity/mode changed')
    def prepare_fresh(self):
        need(os.geteuid()==0,'Observer must be root')
        need(self.path.is_absolute(),'FIFO path must be absolute')
        for ancestor in self.path.parents:
            a=ancestor.lstat();need(stat.S_ISDIR(a.st_mode) and not stat.S_ISLNK(a.st_mode) and not(a.st_mode&0o022),'Unsafe FIFO ancestor')
        s=self.path.parent.lstat();need(stat.S_ISDIR(s.st_mode) and s.st_uid==1001 and stat.S_IMODE(s.st_mode)==0o700,'FIFO parent must be UID1001mode0700')
        if self.path.exists()or self.path.is_symlink():
            s=self.path.lstat();need(stat.S_ISFIFO(s.st_mode),'Existing IPC not plain FIFO');self.target=(s.st_dev,s.st_ino)
            rows=self.corroborate('initial-old-inode',())
            need(not any((f['dev'],f['ino'])==self.target for r in rows for f in r['fds']),'Old FIFO still open')
            old=self.out/'preserved-original-fifo';need(not old.exists() and not old.is_symlink(),'Old FIFO preservation collision')
            os.rename(self.path,old)
            self.old=old;self.setup_phase='ORIGINAL_MOVED'
        else:self.corroborate('initial-no-fifo',())
        # Install only after configuration: failures leave the canonical path absent.
        # Keep original identity separate; target refers only to an installed FIFO.
        self.preserved_target=self.target;self.target=None;self.partial_setup=True
        self.pending=self.out/'pending-fresh-fifo';self.pending_target=None
        need(not self.pending.exists() and not self.pending.is_symlink(),'Fresh FIFO staging collision')
        os.mkfifo(self.pending,0o600)
        s=self.pending.lstat();need(stat.S_ISFIFO(s.st_mode),'Staged FIFO type changed');self.pending_target=(s.st_dev,s.st_ino)
        os.chown(self.pending,1001,1001);os.chmod(self.pending,0o600)
        s=self.pending.lstat();need(stat.S_ISFIFO(s.st_mode) and (s.st_dev,s.st_ino)==self.pending_target and s.st_uid==1001 and stat.S_IMODE(s.st_mode)==0o600,'Staged FIFO identity/mode changed')
        need(not self.path.exists() and not self.path.is_symlink(),'Foreign canonical FIFO appeared')
        os.rename(self.pending,self.path);self.target=self.pending_target;self.partial_setup=False;self.setup_phase='FRESH_INSTALLED'
        self.check_inode();self.event('fresh-inode',dev=s.st_dev,ino=s.st_ino)
        rows=self.corroborate('fresh-no-opener',())
        need(not any((f['dev'],f['ino'])==self.target for r in rows for f in r['fds']),'Fresh FIFO already opened')
        return self.target
