def filesystem_failure(error,stage,roots):
    failure=StageFailure('Candidate filesystem staging failed; inspect stage and errno')
    failure.stage=stage;failure.os_errno=error.errno;failure.os_error_type=type(error).__name__
    failure.controlled_filename='outside-reviewed-staging-paths'
    if isinstance(error.filename,(str,bytes)):
        path=Path(os.fsdecode(error.filename)).absolute()
        for label,root in roots.items():
            root=Path(root).absolute()
            if contains(root,path):
                relative=path.relative_to(root).as_posix()
                if len(relative)<=512 and not any(ord(c)<32 for c in relative):failure.controlled_filename=label+'/'+relative
                break
    return failure


@contextmanager
def filesystem_step(stage,roots):
    try:yield
    except OSError as error:raise filesystem_failure(error,stage,roots) from None


def failure_record(error):
    record=dict(status='candidate_launch_refused_or_stopped',error=type(error).__name__,
                reason=str(error) if isinstance(error,Refusal) else 'External OS, parse or dependency failure; inspect the read-only preflight report',
                native_authority_assigned=False,native_accepted=False)
    if isinstance(error,StageFailure):
        record.update(stage=error.stage,filename=error.controlled_filename,errno=error.os_errno,os_error_type=error.os_error_type)
        if hasattr(error,'probe_metadata'):record['probe_metadata']=error.probe_metadata
    return record


def digest(path):
    with Path(path).open('rb') as stream:return hashlib.file_digest(stream,'sha256').hexdigest()

def tree_manifest(directory):
    root=Path(directory);records={}
    if root.is_symlink() or not root.is_dir():raise Refusal('Original must be a real directory')
    for p in sorted(root.rglob('*')):
        info=p.lstat()
        if stat.S_ISDIR(info.st_mode):continue
        if not stat.S_ISREG(info.st_mode):raise Refusal('Original contains symlink or special file')
        records[p.relative_to(root).as_posix()]=digest(p)
    return records

def verify_private_state(directory,uid,gid):
    """Check actual rootfs metadata; never repair privacy by widening access."""
    directory=Path(directory)
    for path in [directory,*directory.rglob('*')]:
        info=path.lstat()
        if stat.S_ISLNK(info.st_mode) or (info.st_uid,info.st_gid)!=(uid,gid):
            raise Refusal('Candidate private state ownership or symlink check failed')
        expected=0o700 if stat.S_ISDIR(info.st_mode) else 0o600
        if not (stat.S_ISDIR(info.st_mode) or stat.S_ISREG(info.st_mode)) or stat.S_IMODE(info.st_mode)!=expected:
            raise Refusal('Candidate private state permissions must be exactly 0700 directories and 0600 files')

def owns_listener(pid,port,operation=None):
    def mark(value):
        if operation is not None:operation(value)
    inodes=set()
    mark('tcp-table-read')
    for row in Path('/proc/net/tcp').read_text().splitlines()[1:]:
        parts=row.split()
        if parts[1]==f'0100007F:{port:04X}' and parts[3]=='0A':inodes.add(parts[9])
    mark('fd-directory-list')
    descriptors=list((Path('/proc')/str(pid)/'fd').iterdir())
    for fd in descriptors:
        try:
            value=os.readlink(fd)
            if value.startswith('socket:[') and value[8:-1] in inodes:return True
        except (FileNotFoundError,PermissionError):pass
    return False


def owned_listener_probe(pid,ticks,port,user):
    """Inspect only this child's sockets as its UID, without ptrace privileges."""
    def exact_child():
        value=identity(pid)
        return value['uid']==user.pw_uid and value['start_ticks']==ticks
    if not exact_child():raise Refusal('Candidate listener process identity changed before ownership probe')
    read_fd,write_fd=os.pipe();worker=os.fork()
    if worker==0:
        os.close(read_fd)
        stage='drop-worker-uid'
        def operation(value):
            nonlocal stage
            stage=value
        try:
            os.initgroups(user.pw_name,user.pw_gid);os.setgid(user.pw_gid);os.setuid(user.pw_uid)
            os.environ.clear()
            stage='exact-child-start'

def load_token(path,uid):
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW)
    with os.fdopen(fd,'rb') as stream:
        info=os.fstat(stream.fileno())
        if not stat.S_ISREG(info.st_mode) or info.st_uid!=uid or info.st_mode&0o077 or not 12<=info.st_size<=4096:
            raise Refusal('Protected existing operator credential required')
        token=stream.read(4097).decode().strip()
    if len(token)<12:raise Refusal('Protected operator credential is too short')
    return token


def terminate_owned(children):
    for process,ticks in children:
        try:
            if process.poll() is None and (ticks is None or identity(process.pid)['start_ticks']==ticks):process.terminate()
        except (OSError,ValueError,IndexError):pass
    for process,_ in children:
        try:process.wait(timeout=12)
        except subprocess.TimeoutExpired:pass  # Never escalate to killing a reused or unrelated PID.
