def scan_one(pid,proc=Path('/proc')):
    before=identity(pid,proc);p=proc/str(pid);fds=[]
    cwd=path_reference(p/'cwd');root=path_reference(p/'root')
    mount_namespace=os.readlink(p/'ns'/'mnt');mountinfo=(p/'mountinfo').read_text()

    with os.scandir(p/'fd') as entries:
        for entry in entries:
            ref=path_reference(Path(entry.path));s=entry.stat();need((s.st_dev,s.st_ino)==(ref['dev'],ref['ino']),'FD target raced'); lines=(p/'fdinfo'/entry.name).read_text().splitlines()
            flags=[int(x.split()[1],8) for x in lines if x.startswith('flags:')]
            need(len(flags)==1,'FD flags absent/ambiguous')
            fds.append({'fd':int(entry.name),'dev':s.st_dev,'ino':s.st_ino,'flags':flags[0],'socket':stat.S_ISSOCK(s.st_mode),'target':ref['target'],'mode':ref['mode']})
    after=identity(pid,proc);need(stable(before,after),'PID/credential changed during census')

    need(stable(after,identity(pid,proc)),'Identity changed while reading executable')

    need(mount_namespace==os.readlink(p/'ns'/'mnt') and mountinfo==(p/'mountinfo').read_text(),'Mount namespace/table changed during census')
    return {'identity':after,'fds':fds,'exe':exe,'exe_sha256':exe_hash,'cwd':cwd,'root':root,'mount_namespace':mount_namespace,'mountinfo':mountinfo}

def bounded_helper(cmd,timeout=10,validate_json=False):
    global DIAGNOSTIC_BYTES
    import selectors
    p=subprocess.Popen(cmd,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,close_fds=True,cwd='/')
    info={'helper_pid':p.pid,'argv':cmd,'started_ns':time.monotonic_ns(),'timeout':False,'truncated':False,'read_error':None}
    buffers={'stdout':bytearray(),'stderr':bytearray()};sel=selectors.DefaultSelector();deadline=time.monotonic()+timeout
    try:
        try:info['helper_identity']=identity(p.pid)
        except BaseException as e:info['helper_identity_error']=repr(e)
        for name in buffers:
            stream=getattr(p,name);os.set_blocking(stream.fileno(),False);sel.register(stream,selectors.EVENT_READ,name)
        while sel.get_map():
            remaining=deadline-time.monotonic()
            if remaining<=0:info['timeout']=True;break
            for key,_ in sel.select(min(.05,remaining)):
                chunk=os.read(key.fd,8192)
                if not chunk:sel.unregister(key.fileobj);continue
                name=key.data;room=65536-len(buffers[name])
                keep=chunk[:max(0,room)];buffers[name].extend(keep)
                if len(keep)<len(chunk):info['truncated']=True;break
            if info['truncated']:break
    except BaseException as e:info['read_error']=repr(e)
    finally:
        sel.close()
        # Exact Popen child only; never leave helper running when diagnostics fill.
        try:
            if info['timeout'] or info['truncated'] or info['read_error']:
                if p.poll() is None:p.kill()
            try:info['returncode']=p.wait(timeout=max(.01,deadline-time.monotonic()))
            except subprocess.TimeoutExpired:
                info['timeout']=True;p.kill();info['returncode']=p.wait(timeout=2)
        finally:
            p.stdout.close();p.stderr.close()
    info['finished_ns']=time.monotonic_ns()
    if validate_json and not(info['timeout'] or info['truncated'] or info['read_error']) and info['returncode']==0:
        try:json.loads(buffers['stdout'])
        except (ValueError,UnicodeError) as e:info['read_error']='Invalid complete helper JSON: '+repr(e)
    failed=bool(info['timeout'] or info['truncated'] or info['read_error'] or info['returncode']!=0)
    info['streams']={}
    for name,data in buffers.items():
        entry={'bytes':len(data),'sha256':sha_bytes(data)}
        if failed:
            retained=data[:max(0,1048576-DIAGNOSTIC_BYTES)];DIAGNOSTIC_BYTES+=len(retained)
            entry.update(retained_bytes=len(retained),retained_sha256=sha_bytes(retained),hex=retained.hex())
            if len(retained)!=len(data):info['truncated']=True
        info['streams'][name]=entry
    info['complete']=not(info['timeout'] or info['truncated'] or info['read_error'])
    diagnostic(info)
    return bytes(buffers['stdout']),info

def inspect_one(pid,proc=Path('/proc'),*,executable_only=False):
    """Complete FD+executable inspection, with bounded credential fallback everywhere."""
    with FD_LOCK:
        before=identity(pid,proc)
        try:row=executable_one(pid,proc) if executable_only else scan_one(pid,proc)
        except PermissionError:
            need(proc==Path('/proc'),'Credential fallback unavailable for synthetic proc')
            need(len(set(before['uids']))==1 and len(set(before['gids']))==1,'Mixed target credentials')
            cmd=['/usr/bin/setpriv','--no-new-privs','--reuid='+before['uids'][0],'--regid='+before['gids'][0],
                 ('--groups='+','.join(before['groups']) if before['groups'] else '--clear-groups'),sys.executable,'-I',str(Path(__file__).resolve()),('--executable-one' if executable_only else '--census-one'),str(pid)]
            # run() waits/reaps on return and kills/waits on timeout; no helper survives.
            try:raw,result=bounded_helper(cmd,validate_json=True)
            except BaseException as e:
                diagnostic({'target_before':before,'mode':'executable' if executable_only else 'full-census','helper_error':repr(e)})
                raise
            try:after=identity(pid,proc)
            except BaseException as e:after={'inspection_error':repr(e)}
            diagnostic({'target_before':before,'target_after':after,'mode':'executable' if executable_only else 'full-census','helper_result':{k:v for k,v in result.items() if k!='streams'}})
            need(result['complete'] and result['returncode']==0,'Credentialed FD census failed/incomplete: '+str(pid))
            row=json.loads(raw)
        need(stable(before,row['identity']) and stable(before,identity(pid,proc)),'Inspection helper identity mismatch')
        return row

def census(proc=Path('/proc')):
    with FD_LOCK:return _census(proc)

def _census(proc):
    result=[];initial={int(p.name)for p in proc.iterdir()if p.name.isdigit()}
    for pid in sorted(initial):
        try:result.append(inspect_one(pid,proc))
        except FileNotFoundError:
            need(not (proc/str(pid)).exists(),'Live process lost FD/identity: incomplete census')
    final={int(p.name)for p in proc.iterdir()if p.name.isdigit()}
    need(not(final-initial),'New process appeared during census; unaccounted launch')
    return result

def inspect_frames(directory,reference,window):
    """Independent on-disk journal/frame readback; safe only after collector closes."""
    directory=Path(directory); journal=directory/'frames.jsonl'
    rows=[json.loads(x) for x in journal.read_text().splitlines()]
    need(rows,'No captured frames'); seen=set();inodes=set();previous=None;last_time=None
    for ordinal,r in enumerate(rows,1):
        need(r['ordinal']==ordinal,'Journal ordinal gap')
        name=r['saved_frame'];need(name==f'frame-{ordinal:06d}-seq{r["seq"]}.jpg','Unexpected frame name')
        need(name not in seen,'Repeated saved identity');seen.add(name);p=directory/name;s=p.lstat()
        need(stat.S_ISREG(s.st_mode) and s.st_nlink==1,'Non-independent frame file')
        need((s.st_dev,s.st_ino)not in inodes,'Shared frame inode');inodes.add((s.st_dev,s.st_ino))
        b=p.read_bytes();need(len(b)==r['jpeg_bytes'] and sha_bytes(b)==r['sha256'],'Frame readback mismatch')
        ok,seq,why=reference.check_contract(b);need(ok and seq==r['seq'],'Readback JPEG/seq invalid: '+why)
        if previous is not None:need(seq==previous+1,'Unexplained sequence gap/regression')
        previous=seq;t=r['arrival_monotonic_ns']
        need(window['start_ns']<=t<window['deadline_ns'],'Frame outside capture window')
        need(last_time is None or t>=last_time,'Arrival time regression');last_time=t
    need({x.name for x in directory.glob('*.jpg')}==seen,'Extra/unlisted JPEG')
    need(len(rows)==window['frames'] and window['save_skipped']==0,'Incomplete saved frames')
    need(window['end_ns']>=window['deadline_ns'],'Early capture termination')
    need(window['deadline_ns']-window['start_ns']==15_000_000_000,'Not full15s capture')
    # byte bound includes ALL FIFO read bytes (including pre-window) + journal bytes.
    need(journal.stat().st_size==window['journal_bytes'],'Journal byte accounting mismatch')
    need(sum(r['jpeg_bytes']for r in rows)<=window['input_bytes'],'Saved JPEG bytes exceed FIFO input')
    need(window['input_bytes']+journal.stat().st_size<512*1024*1024,'Byte bound reached')
    need(len(rows)<1000,'Frame bound reached')
    arrivals=[r['arrival_monotonic_ns'] for r in rows]
    gaps=[arrivals[0]-window['start_ns'],window['deadline_ns']-arrivals[-1]]+[b-a for a,b in zip(arrivals,arrivals[1:])]
    need(max(gaps)<=window['max_gap_ns'],'Unexplained leading/interframe/trailing timing gap')
    return {'frames':len(rows),'input_plus_journal_bytes':window['input_bytes']+journal.stat().st_size,'max_gap_ns':max(gaps),'journal_sha256':sha_bytes(journal.read_bytes()),'status':'READBACK_PASS'}

class FrameStore:
    """Synchronous complete frame saving; bounds fail, never truncate as success."""
    def __init__(self,directory,reference,max_frames=1000,max_bytes=512*1024*1024):
        self.directory=Path(directory);self.directory.mkdir(mode=0o700);self.ref=reference
        self.dirfd=os.open(self.directory,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC)
        self.journal=open(self.directory/'frames.jsonl','xb',buffering=0)
        self.frames=0;self.input_bytes=0;self.journal_bytes=0;self.last_seq=None;self.max_frames=max_frames;self.max_bytes=max_bytes
    def account(self,n):
        need(n>=0,'Negative input accounting');self.input_bytes+=n
        need(self.input_bytes+self.journal_bytes<self.max_bytes,'BYTE_BOUND_REACHED')
    def append(self,frame,arrival_ns,read_ordinal):
        ok,seq,why=self.ref.check_contract(frame);need(ok and seq is not None and seq>0,'JPEG_CONTRACT: '+why)
        need(self.last_seq is None or seq==self.last_seq+1,'SEQUENCE_GAP_OR_REGRESSION')
        ordinal=self.frames+1;need(ordinal<self.max_frames,'FRAME_BOUND_REACHED')
        name=f'frame-{ordinal:06d}-seq{seq}.jpg'
        row={'ordinal':ordinal,'seq':seq,'arrival_monotonic_ns':arrival_ns,'read_ordinal':read_ordinal,'jpeg_bytes':len(frame),'sha256':sha_bytes(frame),'saved_frame':name}
        b=(json.dumps(row,sort_keys=True)+'\n').encode();need(self.input_bytes+self.journal_bytes+len(b)<self.max_bytes,'BYTE_BOUND_REACHED')
        fd=os.open(name,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW|os.O_CLOEXEC,0o600,dir_fd=self.dirfd)
        with os.fdopen(fd,'wb')as f:f.write(frame);f.flush();os.fsync(f.fileno())
        self.journal.write(b);self.journal_bytes+=len(b);self.frames=ordinal;self.last_seq=seq
    def close(self):
        if not self.journal.closed:os.fsync(self.journal.fileno());self.journal.close()
        if self.dirfd is not None:os.close(self.dirfd);self.dirfd=None

def check_contract(frame: bytes) -> tuple:
    """-> (ok, seq or None, reason) for one JPEG; never raises. Checked: SOI ... EOI, every header segment's length within the frame
    (before the EOI), the JFIF APP0 first, ONE COM "seq=<1..19 digits>" second, no other APPn/COM, exactly one SOF0 (baseline)
    1920x1080, an SOS. The entropy data is the frame cutter's part (any marker but RST/EOI there cuts the frame)."""
    if len(frame) < 6 or frame[:2] != b'\xff\xd8' or frame[-2:] != b'\xff\xd9':
        return False, None, 'not SOI ... EOI'
    pos, segments, seq, sof, sos, end = 2, 0, None, None, False, len(frame) - 2
    while pos + 4 <= end:
        if frame[pos] != 0xFF:
            return False, seq, 'not a marker where one must be'
        marker = frame[pos + 1]
        length = (frame[pos + 2] << 8) | frame[pos + 3]
        if length < 2 or pos + 2 + length > end:
            return False, seq, 'a segment length out of bounds'
        if marker == 0xDA:
            sos = True
            break
        body = frame[pos + 4:pos + 2 + length]
        segments += 1
        if segments == 1 and marker != 0xE0:
            return False, seq, 'APP0 is not the first segment'
        if segments == 1 and not body.startswith(b'JFIF\0'):
            return False, seq, 'APP0 is not JFIF'
        if segments == 2:
            digits = body[len(COM_PREFIX):]
            if marker != 0xFE or not body.startswith(COM_PREFIX) or not 1 <= len(digits) <= 19 or not digits.isdigit():
                return False, seq, 'the second segment is not COM "seq=<n>"'
            seq = int(digits)
        elif segments > 2 and (0xE0 <= marker <= 0xEF or marker == 0xFE):
            return False, seq, 'another APPn/COM segment'
        elif marker in (0xC1, 0xC2, 0xC3, 0xC5, 0xC6, 0xC7, 0xC9, 0xCA, 0xCB, 0xCD, 0xCE, 0xCF):
            return False, seq, 'not baseline'
        elif marker == 0xC0:
            if sof is not None:
                return False, seq, 'two SOF0'
            if len(body) < 5:
                return False, seq, 'SOF0 too short'
            sof = ((body[3] << 8) | body[4], (body[1] << 8) | body[2])
        pos += 2 + length
    if not sos:
        return False, seq, 'no SOS'
    if sof != (WIDTH, HEIGHT):
        return False, seq, f'SOF0 {sof} is not {WIDTH}x{HEIGHT}'
    return True, seq, ''


class Walker:
    """Incremental frame cutter: feed bytes, get whole frames; cuts and garbage resync on the next SOI."""

    def __init__(self):
        self.buffer = bytearray()
        self.cuts = self.garbage = 0

    def drop_partial(self) -> None:
        """The writer closed (EOF): a frame it had not finished can never complete (it never resumes a tail): count it as a cut."""
        if self.buffer:
            self.cuts += 1
            self.buffer.clear()

    def _soi(self, start: int) -> int:
        return self.buffer.find(b'\xff\xd8\xff', start)

    def feed(self, data: bytes) -> list:
        self.buffer += data
        frames = []
        while True:
            start = self._soi(0)
            if start < 0:
                keep = 2 if self.buffer.endswith(b'\xff\xd8') else 1 if self.buffer.endswith(b'\xff') else 0
                if len(self.buffer) > keep:
                    self.garbage += 1
                del self.buffer[:len(self.buffer) - keep]
                return frames
            if start:
                self.garbage += 1
                del self.buffer[:start]
            end, complete = self._walk()
            fresh = self._soi(2)   # a NEW frame's SOI after this one's start (FF D8 FF never occurs inside an encoder's frame)
            if fresh > 0 and (end is None or end > fresh):
                # the writer restarted (it closed on a cut and reopened with no EOF seen in between): this frame can never complete,
                # and a walk that ran past the new SOI would glue two frames together
                self.cuts += 1
                del self.buffer[:fresh]
                continue
            if end is None:
                return frames                    # unfinished: wait for more bytes
            if not complete:                     # a cut frame: resync on the next SOI after this one
                self.cuts += 1
                nxt = self._soi(2)
                del self.buffer[:nxt if nxt > 0 else end]   # (end >= 2: from the cut on, a next SOI still arriving is kept)
                continue
            frames.append(bytes(self.buffer[:end]))
            del self.buffer[:end]

    def _walk(self) -> tuple:
        """-> (end, True) for a whole frame, (cut position, False) for a cut one, (None, False) when more bytes are needed."""
        b, n, pos = self.buffer, len(self.buffer), 2
        while True:
            if pos + 4 > n:
                return None, False
            if b[pos] != 0xFF:
                return pos, False
            marker = b[pos + 1]
            if marker == 0xD8:
                return pos, False
            length = (b[pos + 2] << 8) | b[pos + 3]
            if length < 2:
                return pos, False
            if pos + 2 + length > n:
                return None, False
            pos += 2 + length
            if marker == 0xDA:
                break
        while True:
            hit = b.find(b'\xff', pos)
            if hit < 0 or hit + 1 >= n:
                return None, False
            nxt = b[hit + 1]
            if nxt == 0x00 or 0xD0 <= nxt <= 0xD7:
                pos = hit + 2
                continue
            if nxt == 0xD9:
                return hit + 2, True
            return hit, False                    # another marker (an SOI of the next frame): this one was cut


def percentile(values: list, q: float) -> float:
    if not values:
        return 0.0
    ordered = sorted(values)
    return ordered[min(len(ordered) - 1, int(round(q * (len(ordered) - 1))))]


def open_save_dir(directory: Path) -> int:
    """-> a descriptor of the --save directory (created 0700 when absent), opened O_DIRECTORY|O_NOFOLLOW: every frame is then written
    through this held descriptor, so the path cannot be swapped for a link afterwards. Raises OSError for anything else."""
    try:
        directory.mkdir(mode=0o700, parents=True)
    except FileExistsError:
        pass   # an existing directory is used; anything else fails the O_DIRECTORY open below
    return os.open(directory, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC)


def save_frame(dir_fd: int, name: str, frame: bytes) -> str | None:
    """Write one frame in the held directory without following a link and without replacing anything (-> name, or None if it exists)."""
    try:
        fd = os.open(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, 0o644, dir_fd=dir_fd)
    except FileExistsError:
        return None
    with os.fdopen(fd, 'wb') as handle:
        handle.write(frame)
    return name


def probe(path: Path, seconds: float, contract: str, save: tuple | None, save_count: int, force: bool) -> dict:
    """`save` = (the held --save directory descriptor, its path for the report) or None."""
    try:   # open first (never through a link), then check what was opened: no window between the check and the open
        fd = os.open(path, os.O_RDONLY | os.O_NONBLOCK | os.O_CLOEXEC | os.O_NOFOLLOW)
    except OSError as exc:
        if exc.errno == errno.ENOENT:
            raise
        raise SystemExit(f'refused: {path} cannot be opened as a FIFO ({os.strerror(exc.errno)}; a link is never followed)')
    try:
        info = os.fstat(fd)
        if not stat.S_ISFIFO(info.st_mode):
            raise SystemExit(f'refused: {path} is not a FIFO (this tool never creates or replaces it)')
        if not force:
            readers, unchecked = other_readers(path, target=(info.st_dev, info.st_ino))
            if readers:
                raise SystemExit(f'refused: other readers hold {path} (pids {readers}); a second reader would split frames (--force to skip)')
            if unchecked:
                raise SystemExit(f'refused: {unchecked} processes are not visible to this user, so another reader could be missed (run it as '
                                 'root, or --force when you know nobody else reads the FIFO)')
    except BaseException:
        os.close(fd)
        raise
    walker, frames, seqs, arrivals, failures, saved, save_skipped = Walker(), 0, [], [], {}, [], 0
    read_bytes, last_read, longest_stall, eof_reads, writer_seen, max_per_read = 0, None, 0.0, 0, False, 0
    # The scan above is advisory (a reader may still open the FIFO after it): it is repeated every RESCAN_S s and at the end, and a
    # reader seen at any time voids the run (its numbers then do not count for the go/no-go).
    target, intruders, rescan_at = (info.st_dev, info.st_ino), set(), time.monotonic() + RESCAN_S
    poller = select.poll()
    poller.register(fd, select.POLLIN)
    started = time.monotonic()
    try:
        while time.monotonic() - started < seconds:
            if not force and time.monotonic() >= rescan_at:
                intruders.update(other_readers(path, target=target)[0])
                rescan_at = time.monotonic() + RESCAN_S
            poller.poll(100)
            try:
                data = os.read(fd, CHUNK)
            except OSError as exc:
                if exc.errno in (errno.EAGAIN, errno.EWOULDBLOCK):
                    continue
                raise

                eof_reads += 1
                walker.drop_partial()             # a frame cut by the writer's close: the next writer starts with a fresh SOI
                time.sleep(0.05)
                continue
            writer_seen = True
            if last_read is not None:
                longest_stall = max(longest_stall, now - last_read)
            last_read = now
            read_bytes += len(data)
            whole = walker.feed(data)
            max_per_read = max(max_per_read, len(whole))
            for frame in whole:
                frames += 1
                arrivals.append(now)
                ok, seq, why = check_contract(frame) if contract == 'main' else (True, None, '')
                if not ok:
                    failures[why] = failures.get(why, 0) + 1
                if seq is not None:
                    seqs.append(seq)
                if save is not None and len(saved) < save_count:
                    written = save_frame(save[0], f'frame-{len(saved) + save_skipped:02d}{"-seq%d" % seq if seq is not None else ""}.jpg', frame)
                    if written:
                        saved.append(f'{save[1]}/{written}')
                    else:
                        save_skipped += 1
        if not force:
            intruders.update(other_readers(path, target=target)[0])   # the last look, while this probe still holds its descriptor
    finally:
        os.close(fd)
    ended = time.monotonic()
    elapsed = (arrivals[-1] - arrivals[0]) if len(arrivals) > 1 else 0.0
    window = (ended - arrivals[0]) if arrivals else 0.0   # first frame to the END of the run: a writer that stopped early counts
    if last_read is not None:
        longest_stall = max(longest_stall, ended - last_read)   # the trailing silence too
    gaps = [(b - a) * 1000 for a, b in zip(arrivals, arrivals[1:])]
    distinct = sorted(set(seqs))
    out_of_order = sum(1 for a, b in zip(seqs, seqs[1:]) if b <= a)
    missing = (distinct[-1] - distinct[0] + 1 - len(distinct)) if distinct else 0
    return {'schema': 'sub01.main-fifo-probe.v1', 'fifo': str(path), 'contract': contract, 'seconds': seconds, 'writer_seen': writer_seen,
            'bytes': read_bytes, 'frames': frames, 'contract_failures': failures, 'cuts': walker.cuts, 'garbage_runs': walker.garbage,
            'tail_bytes': len(walker.buffer), 'fps': round((frames - 1) / elapsed, 2) if elapsed > 0 else 0.0,
            'fps_window': round((frames - 1) / window, 2) if window > 0 and frames > 1 else 0.0,
            'trailing_idle_ms': round((ended - arrivals[-1]) * 1000, 2) if arrivals else None,
            'seq': {'first': seqs[0] if seqs else None, 'last': seqs[-1] if seqs else None, 'distinct': len(distinct), 'missing': missing,
                    'out_of_order': out_of_order, 'unique_fps': round((len(distinct) - 1) / elapsed, 2) if elapsed > 0 and distinct else 0.0,
                    'unique_fps_window': round((len(distinct) - 1) / window, 2) if window > 0 and distinct else 0.0},
            # arrival = the time of the read that completed the frame: frames completed by one read share it (max_frames_per_read)
            'arrival_gap_ms': {'p50': round(percentile(gaps, .5), 2), 'p95': round(percentile(gaps, .95), 2), 'p99': round(percentile(gaps, .99), 2),
                               'max': round(max(gaps), 2) if gaps else 0.0},
            'max_frames_per_read': max_per_read, 'longest_read_stall_ms': round(longest_stall * 1000, 2), 'eof_reads': eof_reads, 'saved': saved,
            'save_skipped_existing': save_skipped,
            # the scans can show another reader, never prove there was none (one could open and close between two scans)
            'other_reader_seen': None if force else bool(intruders), 'other_readers_during_run': sorted(intruders)}


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument('--fifo', type=Path, default=Path(DEFAULT_FIFO))
    parser.add_argument('--seconds', type=float, default=30.0)
    parser.add_argument('--contract', choices=('main', 'any'), default='main')
    parser.add_argument('--save', type=Path, help='directory for the first whole frames (.jpg)')
    parser.add_argument('--save-count', type=int, default=3)
    parser.add_argument('--force', action='store_true', help='skip the other-reader check (only when you know nobody else reads it)')
    args = parser.parse_args(argv)
    save = None
    if args.save is not None:
        try:
            save = (open_save_dir(args.save), str(args.save))
        except OSError as exc:
            print(json.dumps({'schema': 'sub01.main-fifo-probe.v1', 'status': 'refused',
                              'reason': f'--save {args.save} is not a usable plain directory ({os.strerror(exc.errno) if exc.errno else exc})'}))
            return 2
    try:
        report = probe(args.fifo, args.seconds, args.contract, save, args.save_count, args.force)
    except SystemExit as stop:
        print(json.dumps({'schema': 'sub01.main-fifo-probe.v1', 'status': 'refused', 'reason': str(stop)}))
        return 2
    except FileNotFoundError:
        print(json.dumps({'schema': 'sub01.main-fifo-probe.v1', 'status': 'refused', 'reason': f'{args.fifo} does not exist'}))
        return 2
    finally:
        if save is not None:
            os.close(save[0])
    report['status'] = 'ok'
    print(json.dumps(report, indent=2))
    return 0


if __name__ == '__main__':
    sys.exit(main())
